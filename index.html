Configuração de Conexão com o Supabase

Insira os dados do seu projeto Supabase (disponíveis em Project Settings > API no painel do Supabase). Seus dados ficarão salvos diretamente no banco de dados na nuvem.

Copie e cole no SQL Editor do Supabase: 
-- 1. TABELA DE ESTAGIÁRIOS
create table if not exists estagiarios (
  id bigint generated by default as identity primary key,
  nome text not null,
  cpf text not null,
  curso text not null,
  faculdade text not null,
  origem text not null,
  setor text not null,
  inicio date not null,
  termino date not null,
  valor_bolsa numeric default 1200.00,
  termo text,
  status text default 'Ativo',
  observacoes text,
  rescisao jsonb default null,
  termo_arquivo jsonb default null,
  folhas_ponto jsonb default '{}'::jsonb,
  pagamentos jsonb default '{}'::jsonb,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- GARANTIR COLUNAS EM CASO DE TABELA JÁ EXISTENTE:
alter table estagiarios add column if not exists termo_arquivo jsonb default null;
alter table estagiarios add column if not exists folhas_ponto jsonb default '{}'::jsonb;
alter table estagiarios add column if not exists pagamentos jsonb default '{}'::jsonb;
alter table estagiarios add column if not exists rescisao jsonb default null;
alter table estagiarios add column if not exists banco text;
alter table estagiarios add column if not exists agencia text;
alter table estagiarios add column if not exists conta text;
alter table estagiarios add column if not exists chave_pix text;

-- Impede CPFs duplicados no banco (rode só depois de remover duplicados existentes, se houver)
-- alter table estagiarios add constraint estagiarios_cpf_unique unique (cpf);

-- 2. TABELA DE MEMORANDOS
create table if not exists memorandos (
  id bigint generated by default as identity primary key,
  numero text not null,
  data text not null,
  competencia text not null,
  qtd_aptos int default 0,
  valor_total numeric default 0,
  responsavel text,
  conteudo jsonb default '{}'::jsonb,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 3. PERMISSÕES DE ACESSO (RLS)
alter table estagiarios enable row level security;
alter table memorandos enable row level security;

drop policy if exists "Permitir tudo estagiarios" on estagiarios;
drop policy if exists "Permitir tudo memorandos" on memorandos;

create policy "Permitir tudo estagiarios" on estagiarios for all using (true) with check (true);
create policy "Permitir tudo memorandos" on memorandos for all using (true) with check (true);
